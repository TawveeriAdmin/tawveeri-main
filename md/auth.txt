  import { useAuth } from '@/lib/auth';

  function AuthExample() {
    const {
      // ✅ Registration

      // ✅ Login
      signInWithEmail,
      signInWithPhone,
      signInWithOAuth,

      // ✅ Logout
      signOut,

      // ✅ Password Management
      resetPassword,      // Forgot password
      updatePassword,     // Change password

      // ✅ Profile
      updateProfile,

      // ✅ Session
      refreshSession,

      // State
      user,
      session,
      loading,
    } = useAuth();

    // ✅ REGISTER with email
    const register = () => signUp({
      email: 'user@example.com',
      password: 'password123',
      full_name: 'Ahmed Ali',
      preferred_language: 'ar',
    });

    // ✅ REGISTER with phone
    const registerPhone = () => signUp({
      phone: '+966501234567',
      password: 'password123',
      full_name: 'Ahmed Ali',
    });

    // ✅ LOGIN with email
    const login = () => signInWithEmail('user@example.com', 'password123');

    // ✅ LOGIN with phone
    const loginPhone = () => signInWithPhone('+966501234567', 'password123');

    // ✅ LOGIN with OAuth
    const loginGoogle = () => signInWithOAuth('google');

    // ✅ LOGOUT
    const logout = () => signOut();

    // ✅ RESET PASSWORD (forgot password)
    const forgotPassword = () => resetPassword('user@example.com');
    // User receives email with reset link
    // After clicking link, they can set new password

    // ✅ CHANGE PASSWORD (while logged in)
    const changePassword = () => updatePassword('newPassword123');

    return <div>All auth features work!</div>;
  }
